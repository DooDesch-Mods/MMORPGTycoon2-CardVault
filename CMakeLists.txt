cmake_minimum_required(VERSION 3.24)
project(MT2CardVault VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CARDVAULT_ORIGIN "https://mmorpgtycoon2cardvault.vercel.app" CACHE STRING "Exact Card Vault service origin")
add_library(MT2CardVault SHARED src/plugin.cpp src/platform.cpp adapters/mt2/cards.cpp)
target_include_directories(MT2CardVault PRIVATE include third_party)
target_compile_definitions(MT2CardVault PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX CARDVAULT_ORIGIN="${CARDVAULT_ORIGIN}")
target_compile_options(MT2CardVault PRIVATE /W4 /utf-8)
target_link_libraries(MT2CardVault PRIVATE winhttp bcrypt crypt32 shell32 ole32 windowscodecs)
set_target_properties(MT2CardVault PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/package/doodesch.cardvault")
add_custom_command(TARGET MT2CardVault POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/mod.json" "$<TARGET_FILE_DIR:MT2CardVault>/mod.json")
enable_testing()
add_executable(card_names_tests tests/card_names_tests.cpp)
target_include_directories(card_names_tests PRIVATE include)
target_compile_options(card_names_tests PRIVATE /utf-8)
add_test(NAME card_names COMMAND card_names_tests)
